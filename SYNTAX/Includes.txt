Nexa Includes
=============

MODULE INCLUDES (angle brackets)
  #include <std/io>
  #include <std/os>
  #include <std/file>
  #include <std/random>
  #include <std/dll>
  #include <std/time>
  #include <std/thread>
  #include <std/inline>   - Enables inline_cpp! { ... } (raw C++ pasted into generated code; use sparingly)
  - Standard library modules

PACKAGE INCLUDES (angle brackets, non-std)
  #include <pkgname/module>
  - Resolved from .nexa/packages/ (project) and ~/.nexa/packages/ (global)
  - Use nexapkg add <pkg> && nexapkg install to install packages

C/C++ HEADERS (.h, .hpp, .hxx, .hh)
  #include "local.hpp"
  #include <vendor/header.h>

  - Recognized by extension. Quoted headers are opened and re-emitted as #include \"<absolute/path>\"
    so the C++ compiler finds them when Nexa builds from a temp .cpp file.
  - Angle form is copied into the generated .cpp as written (use for SDK/install headers on -I paths).
  - Does not apply to std/* (those stay Nexa modules) or to package includes without a header extension.

FILE INCLUDES (quotes)
  #include "file.nxa"
  #include "lib/utils.nxa"
  #include "../shared.nxa"
  #include "/abs/path.nxa"

  Path rules:
    "file.nxa"     - Same directory as current file
    "sub/file.nxa" - Subdirectory
    "../file.nxa"  - Parent directory
    "/path.nxa"    - Absolute path (Linux/Windows)

  - Paths are relative to the file containing the #include
  - Circular includes are prevented
  - Included files can have their own #includes
