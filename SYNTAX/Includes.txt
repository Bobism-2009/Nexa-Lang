Nexa Includes
=============

MODULE INCLUDES (angle brackets)
  #include <std/io>
  #include <std/os>
  #include <std/file>
  #include <std/random>
  #include <std/dll>
  #include <std/time>
  #include <std/thread>
  #include <std/gfx>      - Pixel window (Win32 / Cocoa / X11 / wasm): gfx.open/plot/get/fill/line/text/image/blit/present/key/mouse
  #include <std/inline>   - Enables inline_cpp! { ... } (raw C++ pasted into generated code; use sparingly)
  - Standard library modules

PACKAGE INCLUDES (angle brackets, non-std)
  #include <pkgname/module>
  - Resolved from .nexa/packages/ (project) then ~/.nexa/packages/ (global)
  - Install with: nexapkg install <pkg>   (or: nexapkg add <pkg> && nexapkg install)
  - Pin a version: nexapkg add user/repo@v1.2.0
  - See SYNTAX/Modules.txt (PACKAGES) for the full nexapkg command reference

C/C++ HEADERS (.h, .hpp, .hxx, .hh)
  #include "local.hpp"
  #include <vendor/header.h>
  #include <string.h>
  #include <cstring>

  - Recognized by extension, or by C++ std names with no extension (cstring, cstdio, …).
  - Quoted headers are opened and re-emitted as #include \"<absolute/path>\"
    so the C++ compiler finds them when Nexa builds from a temp .cpp file.
  - Angle form is copied into the generated .cpp as written (use for SDK/install headers on -I paths).
  - Functions, namespaces (`ns::foo`), methods, and types from the header can be
    used as Nexa (no extern fn, no inline_cpp). See SYNTAX/Core.txt (C / C++ HEADERS).
  - Does not apply to std/* (those stay Nexa modules) or to package includes without a header extension.

FILE INCLUDES (quotes)
  #include "file.nxa"
  #include "lib/utils.nxa"
  #include "../shared.nxa"
  #include "/abs/path.nxa"

  Path rules:
    "file.nxa"     - Same directory as current file
    "sub/file.nxa" - Subdirectory
    "../file.nxa"  - Parent directory
    "/path.nxa"    - Absolute path (Linux/Windows)

  - Paths are relative to the file containing the #include
  - Circular includes are prevented
  - Included files can have their own #includes
